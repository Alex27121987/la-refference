// 1. Voir la valeur du paramètre class
const classKey = new URLSearchParams(window.location.search).get('class');
console.log('classKey:', classKey);

// 2. Voir les élèves présents pour ce classKey
const students = JSON.parse(localStorage.getItem('cs_la_reference_students') || '[]');
console.log('élèves trouvés:', students.filter(s => s.classKey === classKey));

// 3. Voir tous les classKey existants
console.log('Tous les classKey élèves:', [...new Set(students.map(s => s.classKey))]);
